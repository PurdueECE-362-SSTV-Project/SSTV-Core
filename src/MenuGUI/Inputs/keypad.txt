void keypad_init_pins() {
    //printf("entered keypad_init_pins\n");

    // Setting Pins 6, 7, 8, and 9 outputs
    gpio_init_dir_fn((1u << 6) | (1u << 7) | (1u << 8) | (1u << 9), GPIO_OUT, GPIO_FUNC_SIO);
    // 2, 3, 4, and 5 as inputs
    gpio_init_dir_fn((1u << 2) | (1u << 3) | (1u << 4) | (1u << 5), GPIO_IN, GPIO_FUNC_SIO);

    col = 0;
    sio_hw->gpio_clr = (1u << 7) | (1u << 8) | (1u << 9);
    sio_hw->gpio_set = (1u << 6);
}

void keypad_init_timer() {
    //printf("entered keypad_init_timer\n");

    // Enable Alarm 0, 1 on TIMER0
    timer_hw->inte |= (1 << ALARM_NUM_DRIVE);

    // Set ISR for IRQ
    irq_set_exclusive_handler(ALARM_IRQ_DRIVE, (irq_handler_t) keypad_drive_column);

    // First INT of time
    timer_hw->alarm[ALARM_NUM_DRIVE] = timer_hw->timerawl + SECOND;

    timer_hw->inte |= (1 << ALARM_NUM_READ);
    timer_hw->alarm[ALARM_NUM_READ] = timer_hw->timerawl + SECOND_1;
    irq_set_exclusive_handler(ALARM_IRQ_READ, (irq_handler_t) keypad_isr);

    // Enable IRQ
    irq_set_enabled(ALARM_IRQ_DRIVE, true);
    irq_set_enabled(ALARM_IRQ_READ, true);
}

void keypad_drive_column() {
    //printf("entered keypad_drive_column\n");

    // Ack interrupt and reset alarm 0
    timer_hw->intr |= (1 << ALARM_NUM_DRIVE);

    //print_current_gpio_state();

    uint prev_col = col;
    col++;
    if (col > 3) col = 0;

    // Reset then Set respective column - flips all the bits (requires one of them to be 1 initially (gpio[6] in this case))
    sio_hw->gpio_togl = (1u << (6 + col)) | (1u << (6 + prev_col));
    // sio_hw->gpio_clr = (1u << 6) | (1u << 7) | (1u << 8) | (1u << 9);
    // sio_hw->gpio_set = (1u << ((3 - col) + 6));
    timer_hw->alarm[ALARM_NUM_DRIVE] = timer_hw->timerawl + DELAY_DRIVE_READ;
}

uint8_t keypad_read_rows() {
    //printf("entered keypad_read_rows\n");
    // Take gpio_in and mask relevant values
    // uint8_t mask = (1 << 5) | (1 << 4) | (1 << 3) | (1 << 2);
    // uint8_t output = (uint8_t) ((sio_hw->gpio_in & mask) >> 2);

    // OFFSET back by to acc for LSB being at [2]
    return (uint8_t) ((sio_hw->gpio_in & ((1 << 5) | (1 << 4) | (1 << 3) | (1 << 2))) >> 2);
}

// Key keymap index for a specific row and (global) column
uint8_t get_keymap_index (uint8_t row) {
    return (col * 4 + (row - 1));
}

bool keypress (uint row) {
    if (keypad_read_rows() > 0) print_uint32_to_binary(keypad_read_rows());
    return keypad_read_rows() & (1 << (row - 1));
}

void keypad_isr() {
    // Ack interrupt and reset alarm 1
    timer_hw->intr |= (1u << ALARM_NUM_READ);

    int row;
    uint8_t key_mapped_index;
    bool key_press;
    uint16_t key_event;

    // Inline for each row:
    // GP2
    for (row = 4; row >= 1; row --) {
        // Inline get_keymap_index:
        key_mapped_index = get_keymap_index(row);
        key_press = keypress(row);
        if (key_press != (state[key_mapped_index])) {
            state[key_mapped_index] = key_press;
            //printf("Pressed: %c col %d\n", keymap[key_mapped_index], col);
            key_event = (uint16_t) ((key_press << 8) | (uint8_t) keymap[key_mapped_index]);
            key_push(key_event);
        }
    }
    timer_hw->alarm[ALARM_NUM_READ] = timer_hw->timerawl + DELAY_DRIVE_READ;

}

void keypad_isr() {
    int row;

    // Print Keypad values
    if (gpio_get_irq_event_mask(2) & GPIO_IRQ_EDGE_RISE) {
        gpio_acknowledge_irq(2, GPIO_IRQ_EDGE_RISE);
        row = 4;
    }

    if (gpio_get_irq_event_mask(3) & GPIO_IRQ_EDGE_RISE) {
        gpio_acknowledge_irq(3, GPIO_IRQ_EDGE_RISE);
        row = 3;
    }

    if (gpio_get_irq_event_mask(4) & GPIO_IRQ_EDGE_RISE) {
        gpio_acknowledge_irq(4, GPIO_IRQ_EDGE_RISE);
        row = 2;
    }

    if (gpio_get_irq_event_mask(5) & GPIO_IRQ_EDGE_RISE) {
        gpio_acknowledge_irq(5, GPIO_IRQ_EDGE_RISE);
        row = 1;
    }
    char key = (&keymap[4 - (row)])[4 * (4 - (col + 1))];
    printf("%c\n", key);
}

void init_keypad_irq() {
    gpio_add_raw_irq_handler_masked((1u << 2) | (1u << 3) | (1u << 4) | (1u << 5), (irq_handler_t) &keypad_isr);

    // Enabling IRQ for GPIO
    uint i = 2;

    for (i = 2; i < 6; i++) {
        gpio_set_irq_enabled(i, GPIO_IRQ_EDGE_RISE, true);
    }

    // IRQ set IO Bank  
    irq_set_enabled(IO_IRQ_BANK0, true);
}
