#include <stdlib.h>
#include <stdio.h>
#include <stdint.h>

#include "DataIO.h"

static float* bufferGenFloat(uint16_t size) {
    float *output = malloc(size * sizeof(float));
    if (!output) return NULL;
    return output;
}

void initializeQueueNode (struct linked_queue_t* queue_node) {
    queue_node->next_node = NULL;
    queue_node->buffer_adcOut = NULL;
    queue_node->buffer_fftOut = NULL;
    queue_node->prev_node = NULL;
}

void queue_init (uint8_t SIZE) {
    struct linked_queue_t* first_node_queue;
    struct linked_queue_t* prev_node_queue;
    initializeQueueNode(prev_node_queue);

    // Intialiing prev queue
    prev_node_queue->next_node = malloc(sizeof(struct linked_queue_t*));
    if (!(prev_node_queue->next_node)) printf("Failed to allocate queue node\n");

    // Making a fresh SIZE node(s) long queue with fresh buffers
    for (uint8_t iter = 0; iter < SIZE; iter++) {
        struct linked_queue_t* new_node_queue = prev_node_queue->next_node;


        ////    ////    ////    ////    ////    
        // Assign Prev Queue to the prev - initiaize first node to attach last note to at the end //
        if (iter == 0) {
            first_node_queue = new_node_queue;
            new_node_queue->prev_node = 0;
        }

        else new_node_queue->prev_node = prev_node_queue;
        ////    ////    ////    ////    ////    


        ////    ////    ////    ////    ////    
        // Assign buffers to the queue //
        // ADC Buffer
        new_node_queue->buffer_adcOut = bufferGenFloat(BUFFER_SIZE);
        if (!(new_node_queue->buffer_adcOut)) printf("Failed to allocate Buffer\n");

        // FFT Buffer
        new_node_queue->buffer_fftOut = bufferGenFloat(BUFFER_SIZE);
        if (!(new_node_queue->buffer_fftOut)) printf("Failed to allocate Buffer\n");
        ////    ////    ////    ////    ////    


        ////    ////    ////    ////    ////  
        // Malloc the next queue to write on - or assign the first queue to the last node (loop)
        if (iter < (SIZE - 1)) {
            new_node_queue->next_node = malloc(sizeof(struct linked_queue_t*));
            if (!(new_node_queue->next_node)) printf("Failed to allocate queue node\n");
        }

        // Recognizing Last node to assign to 'prev' of first - 'next' of last
        else {
            first_node_queue->prev_node = new_node_queue;
            new_node_queue->next_node = first_node_queue;
        }
        ////    ////    ////    ////    ////

        // Update prev queue to new queue for next update
        prev_node_queue = new_node_queue;
    }
}
